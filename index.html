function fillAttach(id, series) { const sel = document.getElementById(id); sel.innerHTML = ""; // Определяем, для какой стороны заполняем // id="attach-a" это сторона B (шток) - левая колонка // id="attach-b" это сторона A (трубка) - правая колонка const isLeftSide = id === 'attach-a'; // true для левой колонки (шток) // Добавляем заголовок "ПОДСОЕДИНЕНИЯ" (disabled, чтобы нельзя было выбрать) const headerOption = document.createElement('option'); headerOption.value = ""; headerOption.text = "══════ ПОДСОЕДИНЕНИЯ ══════"; headerOption.disabled = true; headerOption.selected = true; // Делаем его выбранным по умолчанию sel.appendChild(headerOption); // Добавляем разделитель (опционально) const separatorOption = document.createElement('option'); separatorOption.value = ""; separatorOption.text = "──────────────────────"; separatorOption.disabled = true; sel.appendChild(separatorOption); // Фильтруем и добавляем все наконечники attachments.filter(a => { // Проверяем, подходит ли по серии if (!a.series.includes(series)) return false; // Для резьбы проверяем соответствие стороне if (a.id.startsWith('thread-')) { // Для левой колонки (шток) нужны id с суффиксом -b // Для правой колонки (трубка) нужны id с суффиксом -a return isLeftSide ? a.id.endsWith('-b') : a.id.endsWith('-a'); } // Для всех остальных наконечников (не резьба) - показываем везде return true; }).forEach(a => { sel.add(new Option(`${a.name} (+${a.dist}мм)`, `${a.id}|${a.dist}`)); }); // Если после фильтрации ничего не добавилось if (sel.options.length <= 2) { // только заголовок и разделитель sel.add(new Option("Нет доступных наконечников", "none|0")); } }
